# 创建 Gateway IAM 角色
assume_role_policy = {
    "Version": "2012-10-17",
    "Statement": [{
        "Effect": "Allow",
        "Principal": {
            "Service": "bedrock-agentcore.amazonaws.com"
        },
        "Action": "sts:AssumeRole"
    }]
}

iam_client.create_role(
    RoleName="gateway-demo-role",
    AssumeRolePolicyDocument=json.dumps(assume_role_policy)
)

创建示例 Lambda 函数
待执行
这是什么? 一个模拟订单管理的 Lambda 函数。 它接收工具调用请求，根据 tool_name 执行不同的业务逻辑。
Python (Lambda Handler) 
def lambda_handler(event, context):
    """订单管理 Lambda - 被 Gateway 调用"""

    # Gateway 传入的工具名称和参数
    tool_name = event.get('name', '')
    arguments = event.get('arguments', {})

    if 'get_order' in tool_name:
        order_id = arguments.get('orderId')
        return {
            'statusCode': 200,
            'body': json.dumps({
                'orderId': order_id,
                'status': 'shipped',
                'items': ['Item A', 'Item B'],
                'total': 99.99
            })
        }
    elif 'update_order' in tool_name:
        # 更新订单逻辑...
        return {'statusCode': 200, 'body': 'Order updated'}

创建 Gateway 并定义 MCP 工具
待执行
核心步骤! 创建 Gateway 并定义工具的 Schema。 这里的 toolSchema 告诉 Agent 这个工具能做什么、需要什么参数。
Python (boto3) 
# 创建 Gateway (MCP 协议)
gateway = gateway_client.create_gateway(
    name="order-management-gateway",
    roleArn=role_arn,
    protocolType='MCP',  # ← 关键: 使用 MCP 协议
    authorizerType='CUSTOM_JWT',
    authorizerConfiguration={
        "customJWTAuthorizer": {
            "allowedClients": [client_id],
            "discoveryUrl": discovery_url
        }
    }
)

# 定义 MCP 工具 Schema
tool_schema = [
    {
        "name": "get_order_tool",
        "description": "查询订单状态和详情",
        "inputSchema": {
            "type": "object",
            "properties": {
                "orderId": {
                    "type": "string",
                    "description": "订单ID"
                }
            },
            "required": ["orderId"]
        }
    }
]

# 添加 Lambda Target
gateway_client.create_gateway_target(
    gatewayIdentifier=gateway['gatewayId'],
    name="LambdaOrderTools",
    targetConfiguration={
        "mcp": {
            "lambda": {
                "lambdaArn": lambda_arn,
                "toolSchema": {"inlinePayload": tool_schema}
            }
        }
    }
)

    实验 2: 将 OpenAPI 规范转换为 MCP 工具
    实验目标: 了解如何将现有的 REST API (OpenAPI 规范) 自动转换为 MCP 工具。 Gateway 会自动解析您的 API 定义并生成对应的工具。
    转换对比
    转换前: OpenAPI 规范
    {
  "openapi": "3.0.0",
  "info": {"title": "Weather API"},
  "paths": {
    "/weather": {
      "get": {
        "operationId": "getWeather",
        "summary": "获取天气信息",
        "parameters": [{
          "name": "city",
          "in": "query",
          "required": true,
          "schema": {"type": "string"}
        }]
      }
    }
  }
}
    转换后: MCP 工具
    {
  "name": "getWeather",
  "description": "获取天气信息",
  "inputSchema": {
    "type": "object",
    "properties": {
      "city": {
        "type": "string",
        "description": "城市名称"
      }
    },
    "required": ["city"]
  }
}

// Agent 现在可以调用:
// tool.getWeather({city: "北京"})
    Gateway 自动完成的转换
    OpenAPI 元素 MCP 工具属性 说明
    operationId name API 操作 ID 变为工具名称
    summary / description description API 描述变为工具描述
    parameters inputSchema 请求参数变为输入 Schema
    securitySchemes Outbound Auth 自动处理 API Key / OAuth
    实现代码
    Python (boto3) 
    # 将 OpenAPI 规范转换为 MCP 工具
gateway_client.create_gateway_target(
    gatewayIdentifier=gateway_id,
    name="WeatherAPITools",
    targetConfiguration={
        "mcp": {
            "openApi": {
                "specSource": {
                    # 方式1: 从 S3 加载
                    "s3": {
                        "bucketName": "my-api-specs",
                        "objectKey": "weather-api.yaml"
                    }
                    # 方式2: 内联定义
                    # "inlineSpec": openapi_json_string
                },
                "actions": [
                    {
                        "path": "/weather",
                        "method": "GET",
                        "toolName": "get_weather",
                        "toolDescription": "获取指定城市的天气信息"
                    }
                ]
            }
        }
    },
    # 出站认证: 调用 API 时使用的凭证
    credentialProviderConfigurations=[
        {"credentialProviderType": "API_KEY"}
    ]
)

    Token 使用 ~50,000 ~2,000 25x 节省
    响应延迟 ~3.2s ~1.1s 3x 提速
    工具选择准确率 ~75% ~95% 20% 提升

    实验 3: MCP 工具调用指南
    完成实验 1、2 或 4 创建 Gateway 后，使用本指南了解如何调用 MCP 工具。
    在 Claude Desktop 的配置文件中添加 Gateway 作为 MCP Server

    配置文件位置:

    • macOS: ~/Library/Application Support/Claude/claude_desktop_config.json
    • Windows: %APPDATA%\Claude\claude_desktop_config.json
    claude_desktop_config.json 
    {
  "mcpServers": {
    "agentcore-gateway": {
      "url": "<GATEWAY_URL>",
      "transport": "streamable-http",
      "headers": {
        "Authorization": "Bearer <ACCESS_TOKEN>"
      }
    }
  }
}
    注意: Access Token 有效期通常为 1 小时。生产环境建议使用 Token 刷新机制或 OAuth2 客户端凭证流。
    Python (MCP Client) 
    import boto3
import httpx

# ========== 1. 配置信息 (从实验步骤获取) ==========
COGNITO_CLIENT_ID = "<CLIENT_ID>"      # 步骤2返回的 Client ID
GATEWAY_URL = "<GATEWAY_URL>"          # 步骤4返回的 Gateway URL

# ========== 2. 获取 Access Token ==========
cognito_client = boto3.client('cognito-idp', region_name='us-west-2')

token_response = cognito_client.initiate_auth(
    ClientId=COGNITO_CLIENT_ID,
    AuthFlow='USER_PASSWORD_AUTH',
    AuthParameters={
        'USERNAME': 'testuser',
        'PASSWORD': 'TestPassword123!'
    }
)
access_token = token_response['AuthenticationResult']['AccessToken']

# ========== 3. 调用 MCP 工具 ==========
headers = {
    "Authorization": f"Bearer {access_token}",
    "Content-Type": "application/json"
}

# 列出所有工具
list_response = httpx.post(GATEWAY_URL, headers=headers, json={
    "jsonrpc": "2.0", "id": 1, "method": "tools/list"
})
print("可用工具:", list_response.json())

# 调用工具 (示例: 订单查询)
call_response = httpx.post(GATEWAY_URL, headers=headers, json={
    "jsonrpc": "2.0",
    "id": 2,
    "method": "tools/call",
    "params": {
        "name": "<TOOL_NAME>",
        "arguments": {"param": "value"}
    }
})
print("工具响应:", call_response.json())
    Gateway 使用 Cognito JWT Token 进行入站认证
    认证信息来源
    信息 来源 说明
    Pool ID 实验步骤 2 返回 Cognito 用户池 ID，格式: us-west-2_xxxxxxxx
    Client ID 实验步骤 2 返回 Cognito App Client ID，格式: 26位字母数字
    Username Demo 自动创建 testuser
    Password Demo 自动创建 TestPassword123!
    Gateway URL 实验步骤 4 返回 MCP Server 端点 URL
    命令行获取 Token
    AWS CLI 
    # 获取 Access Token
aws cognito-idp initiate-auth \
  --client-id "<CLIENT_ID>" \
  --auth-flow USER_PASSWORD_AUTH \
  --auth-parameters USERNAME=testuser,PASSWORD=TestPassword123! \
  --region us-west-2

# 测试调用 Gateway (使用 curl)
curl -X POST "<GATEWAY_URL>" \
  -H "Authorization: Bearer <ACCESS_TOKEN>" \
  -H "Content-Type: application/json" \
  -d '{"jsonrpc":"2.0","id":1,"method":"tools/list"}'
    Token 有效期
    • Access Token: 默认 1 小时
    • Refresh Token: 默认 30 天
    • 生产环境建议实现自动刷新机制
    MCP (Model Context Protocol) 是一种标准化的工具调用协议，基于 JSON-RPC 2.0
    常用方法
    方法 说明 示例参数
    tools/list 列出所有可用工具及其 Schema 无
    tools/call 调用指定工具 {"name": "tool_name", "arguments": {...}}
